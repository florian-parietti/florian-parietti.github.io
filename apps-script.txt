function doGet(e) {
  try {
    var sheet = SpreadsheetApp
      .openById("1N1HyFaykcNsPy5ESyrGHXHYGDGyYmkir81Jscg-AqB4")
      .getSheetByName("Leads");

    sheet.appendRow([
      e.parameter.date       || new Date().toLocaleString("fr-BE"),  // A  Date
      e.parameter.statut     || "Nouveau",                           // B  Statut
      e.parameter.instagram  || "",                                  // C  Instagram
      e.parameter.whatsapp   || "",                                  // D  WhatsApp
      e.parameter.age        || "",                                  // E  Âge
      e.parameter.profession || "",                                  // F  Profession
      e.parameter.pilier     || "",                                  // G  Leviers
      e.parameter.objectif   || "",                                  // H  Objectif
      e.parameter.email      || "",                                  // I  Email
      "",                                                            // J  Relance
      ""                                                             // K  Notes
    ]);

    return ContentService.createTextOutput("OK")
      .setMimeType(ContentService.MimeType.TEXT);
  } catch(err) {
    return ContentService.createTextOutput("Error: " + err.toString())
      .setMimeType(ContentService.MimeType.TEXT);
  }
}

function doPost(e) { return doGet(e); }
