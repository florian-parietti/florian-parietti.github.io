/* ═══════════════════════════════════════════════════════════
   SCRIPT À COLLER DANS APPS SCRIPT
   Il crée les en-têtes automatiquement si la feuille est vide.
   ═══════════════════════════════════════════════════════════ */

var SHEET_ID = "1N1HyFaykcNsPy5ESyrGHXHYGDGyYmkir81Jscg-AqB4";

function doGet(e) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);

    // cherche "Leads", sinon prend la première feuille
    var sheet = ss.getSheetByName("Leads") || ss.getSheets()[0];

    // crée les en-têtes si la feuille est vide
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Date","Statut","Instagram","WhatsApp","Age",
        "Profession","Leviers","Objectif","Email","Relance","Notes"
      ]);
    }

    var p = (e && e.parameter) ? e.parameter : {};

    sheet.appendRow([
      p.date       || new Date().toLocaleString("fr-BE"),
      p.statut     || "Nouveau",
      p.instagram  || "",
      p.whatsapp   || "",
      p.age        || "",
      p.profession || "",
      p.pilier     || "",
      p.objectif   || "",
      p.email      || "",
      "",
      ""
    ]);

    return ContentService.createTextOutput("OK")
      .setMimeType(ContentService.MimeType.TEXT);

  } catch (err) {
    return ContentService.createTextOutput("ERREUR: " + err.toString())
      .setMimeType(ContentService.MimeType.TEXT);
  }
}

function doPost(e) { return doGet(e); }
